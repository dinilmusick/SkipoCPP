cmake_minimum_required(VERSION 3.16)
project(SkipoCPP VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(GutCPP CONFIG REQUIRED)
find_package(KrnlCpp CONFIG REQUIRED)
find_package(KloksCPP CONFIG REQUIRED)
find_package(FrnlCPP CONFIG REQUIRED)
find_package(FloksCPP CONFIG REQUIRED)
add_library(skipocpp INTERFACE)
add_library(SkipoCPP::core ALIAS skipocpp)
set_target_properties(skipocpp PROPERTIES EXPORT_NAME core)
target_compile_features(skipocpp INTERFACE cxx_std_17)
target_include_directories(skipocpp INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(skipocpp INTERFACE gutcpp::gutcpp krnl::krnlcpp KloksCPP::klokscpp FrnlCPP::semantic FloksCPP::core)
install(DIRECTORY include/skipo DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS skipocpp EXPORT SkipoCPPTargets)
install(EXPORT SkipoCPPTargets NAMESPACE SkipoCPP:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SkipoCPP)
configure_package_config_file(cmake/SkipoCPPConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/SkipoCPPConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SkipoCPP)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SkipoCPPConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SkipoCPPConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/SkipoCPPConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SkipoCPP)
include(CTest)
if(BUILD_TESTING)
  foreach(test_name application workflow ui host)
    add_executable(skipocpp_${test_name}_test tests/${test_name}.cpp)
    target_link_libraries(skipocpp_${test_name}_test PRIVATE SkipoCPP::core)
    if(MSVC)
      target_compile_options(skipocpp_${test_name}_test PRIVATE /W4 /WX /permissive- /utf-8 /UNDEBUG)
    else()
      target_compile_options(skipocpp_${test_name}_test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
    endif()
    add_test(NAME skipocpp_${test_name} COMMAND skipocpp_${test_name}_test)
  endforeach()
endif()
