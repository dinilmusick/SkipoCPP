cmake_minimum_required(VERSION 3.16)
project(SkipoCPPExternalConsumer LANGUAGES CXX)
find_package(SkipoCPP CONFIG REQUIRED)
add_executable(skipocpp_external main.cpp)
target_link_libraries(skipocpp_external PRIVATE SkipoCPP::core)
if(MSVC)
  target_compile_options(skipocpp_external PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(skipocpp_external PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME installed_external_consumer COMMAND skipocpp_external)
